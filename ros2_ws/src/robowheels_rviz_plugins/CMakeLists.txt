cmake_minimum_required(VERSION 3.8)
project(robowheels_rviz_plugins)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets Test Network)
foreach(dep rviz_common pluginlib rclcpp geometry_msgs sensor_msgs nav_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/control_panel.cpp include/robowheels_rviz_plugins/control_panel.hpp)
target_include_directories(${PROJECT_NAME} PUBLIC include)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets Qt5::Network)
ament_target_dependencies(${PROJECT_NAME} rviz_common pluginlib rclcpp geometry_msgs sensor_msgs nav_msgs)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
add_executable(control_panel_smoke_test test/control_panel_smoke_test.cpp)
target_include_directories(control_panel_smoke_test PRIVATE include)
target_link_libraries(control_panel_smoke_test ${PROJECT_NAME} Qt5::Widgets Qt5::Test)
ament_target_dependencies(control_panel_smoke_test rviz_common rclcpp geometry_msgs sensor_msgs nav_msgs)
install(TARGETS control_panel_smoke_test DESTINATION lib/${PROJECT_NAME})
ament_package()
